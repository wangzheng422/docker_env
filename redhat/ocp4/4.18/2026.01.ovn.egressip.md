# OVN egressip limitation workaround

## Background

customer install a ocp 4.18 on 3rd part osp, using baremetal install method. and the osp has limitation of 10 elastic ip on each node. but the security require each namespace has a dedicate egress ip, and there are 40+ namespaces. and the elastic ip pools contains different subnets, and worker node has different associated relationship with elastic ip pools subnets. current ovn does not support node selector for egressip. and only single pool of egressip is supported. so we need to implement a workaround to support this requirement.

the basic idea is to use a gateway pod for each namespace to provide egress ip for the namespace. use ovn config to route the traffic to the gateway pod for the namespace pod. and use a tool pod to monitor the gateway pod's ip, and change ovn config's next hop ip when the gateway pod restart.

## Workaround

```bash

oc create ns ns-blue

# 创建 SA 
oc create sa apb-syncer -n ns-blue
oc delete role apb-syncer-role -n ns-blue --ignore-not-found
oc delete rolebinding apb-syncer-binding -n ns-blue --ignore-not-found

# 2. 重新创建 Role (明确定义在 ns-blue 内)
cat <<EOF | oc apply -f -
apiVersion: rbac.authorization.k8s.io/v1
kind: Role
metadata:
  name: apb-syncer-role
  namespace: ns-blue
rules:
- apiGroups: [""]
  resources: ["pods"]
  verbs: ["get", "list", "watch"]
- apiGroups: ["k8s.ovn.org"]
  resources: ["adminpolicybasedexternalroutes"]
  verbs: ["get", "list", "patch", "update"]
EOF

# 3. 手动创建 RoleBinding (将 SA 和 Role 绑定)
cat <<EOF | oc apply -f -
apiVersion: rbac.authorization.k8s.io/v1
kind: RoleBinding
metadata:
  name: apb-syncer-binding
  namespace: ns-blue
subjects:
- kind: ServiceAccount
  name: apb-syncer
  namespace: ns-blue
roleRef:
  kind: Role
  name: apb-syncer-role
  apiGroup: rbac.authorization.k8s.io
EOF


oc label node worker-01-demo egress-node=ns-blue-group


cat <<EOF | oc apply -f -
apiVersion: "k8s.cni.cncf.io/v1"
kind: NetworkAttachmentDefinition
metadata:
  name: ns1-external-macvlan
  namespace: ns-blue  # 必须和网关 Pod 在同一个 NS
spec:
  config: '{
      "cniVersion": "0.3.1",
      "type": "macvlan",
      "master": "br-ex",
      "mode": "bridge",
      "ipam": {
        "type": "static"
      }
    }'
EOF

# 创建 Gateway 专用的 SA
oc create sa gateway-sa -n ns-blue

# 为该 SA 授予 privileged SCC
oc adm policy add-scc-to-user privileged -z gateway-sa -n ns-blue

cat <<EOF | oc apply -f -
apiVersion: apps/v1
kind: Deployment
metadata:
  name: ns-blue-gateway
  namespace: ns-blue
spec:
  replicas: 1 
  selector:
    matchLabels:
      app: ns-blue-gateway
  template:
    metadata:
      labels:
        app: ns-blue-gateway
      annotations:
        k8s.v1.cni.cncf.io/networks: '[
          {
            "name": "ns1-external-macvlan", 
            "interface": "macvlan0",
            "ips": ["192.168.99.2/24"]
          }
        ]'
    spec:
      serviceAccountName: gateway-sa
      nodeSelector:
        egress-node: ns-blue-group
      containers:
      - name: gateway
        image: registry.redhat.io/openshift4/ose-egress-router:latest
        command: ["/bin/sh", "-c"]
        args:
        - |
          # 1. 开启内核转发（这是做网关的前提）
          sysctl -w net.ipv4.ip_forward=1;
          
          # 2. 确保 macvlan0 接口已启动
          ip link set macvlan0 up;
          
          # 3. 设置默认网关，指向物理路由器的 IP
          # 这一步非常重要，否则 Pod 里的包不知道怎么去外网
          ip route add default via 192.168.99.1 dev macvlan0;
          
          # 4. 设置核心 SNAT 规则
          # 意思是：凡是进来的包，只要是从 macvlan0 出去的，全都把源 IP 改成 192.168.99.2
          iptables -t nat -A POSTROUTING -o macvlan0 -j SNAT --to-source 192.168.99.2;
          
          # 5. 为了调试方便，加一条日志
          echo "交大版手动网关已启动！出口 IP: 192.168.99.2，物理网关: 192.168.99.1";
          
          # 6. 防止容器退出
          sleep infinity;
        securityContext:
          privileged: true
EOF

```

```python
# 文件名: controller.py
import os
from kubernetes import client, config, watch

def update_apb(ips):
    api = client.CustomObjectsApi()
    
    # --- 关键修正：将 IP 字符串列表转换为对象列表 ---
    # 原来是: ["10.128.x.y"]
    # 现在要变成: [{"ip": "10.128.x.y"}]
    formatted_ips = [{"ip": ip} for ip in ips]
    # -------------------------------------------

    body = {
        "spec": {
            "nextHops": {
                "static": formatted_ips
            }
        }
    }
    try:
        api.patch_namespaced_custom_object(
            group="k8s.ovn.org",
            version="v1",
            namespace="ns-blue",
            plural="adminpolicybasedexternalroutes",
            name="ns1-route",
            body=body
        )
        print(f"Successfully updated APB 'ns1-route' with: {formatted_ips}")
    except Exception as e:
        print(f"Failed to patch APB: {e}")

def main():
    config.load_incluster_config()
    v1 = client.CoreV1Api()
    w = watch.Watch()
    for event in w.stream(v1.list_namespaced_pod, namespace="ns-blue", label_selector="app=ns-blue-gateway"):
        pods = v1.list_namespaced_pod("ns-blue", label_selector="app=ns-blue-gateway")
        ips = [p.status.pod_ip for p in pods.items if p.status.phase == "Running" and p.status.pod_ip]
        if ips:
            update_apb(ips)

if __name__ == "__main__":
    main()
```

apiVersion: k8s.ovn.org/v1
kind: AdminPolicyBasedExternalRoute
metadata:
  name: ns1-route
  namespace: ns-blue
spec:
  from:
    namespaceSelector:
      matchLabels:
        kubernetes.io/metadata.name: ns-blue
  nextHops:
    static:
    - ip: "192.168.99.1"
EOF
