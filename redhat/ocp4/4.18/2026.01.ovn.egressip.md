# OVN egressip limitation workaround

## Background

customer install a ocp 4.18 on 3rd part osp, using baremetal install method. and the osp has limitation of 10 elastic ip on each node. but the security require each namespace has a dedicate egress ip, and there are 40+ namespaces. and the elastic ip pools contains different subnets, and worker node has different associated relationship with elastic ip pools subnets. current ovn does not support node selector for egressip. and only single pool of egressip is supported. so we need to implement a workaround to support this requirement.

the basic idea is to use a gateway pod for each namespace to provide egress ip for the namespace. use ovn config to route the traffic to the gateway pod for the namespace pod. and use a tool pod to monitor the gateway pod's ip, and change ovn config's next hop ip when the gateway pod restart.

in this demo, we have ocp 4.18 installed, 3 master, 2 worker, worker node ip is 192.168.99.26, 192.168.99.27, I have a big rhel 9 machine, and I create the 5 node with kvm, so there should no mac restrict on the network.

```bash
# 创建 Gateway 专用的 SA
oc create sa gateway-sa -n ns-egress-infra

# 为该 SA 授予 privileged SCC

cat <<'EOF' | oc apply -f -
apiVersion: apps/v1
kind: Deployment
metadata:
  name: ns-blue-gateway
  namespace: ns-egress-infra
spec:
  replicas: 1 
  selector:
    matchLabels:
      app: ns-blue-gateway
  template:
    metadata:
      labels:
        app: ns-blue-gateway
      annotations:
        k8s.v1.cni.cncf.io/networks: '[
          {
            "name": "ns-blue-external-macvlan", 
            "interface": "macvlan0",
            "ips": ["192.168.99.2/24"]
          }
        ]'
    spec:
      serviceAccountName: gateway-sa
      nodeSelector:
        egress-node: ns-blue-group
      containers:
      - name: gateway
        image: registry.redhat.io/openshift4/ose-egress-router:latest
        command: ["/bin/sh", "-c"]
        args:
        - |
          # 0. 开启 IP 转发 & 关闭反向路径过滤 (rp_filter) !!! 关键 !!!
          # rp_filter=1 (Strict) 会丢弃源 IP 不符合路由表预期的包
          sysctl -w net.ipv4.ip_forward=1
          sysctl -w net.ipv4.conf.all.rp_filter=0
          sysctl -w net.ipv4.conf.default.rp_filter=0
          sysctl -w net.ipv4.conf.eth0.rp_filter=0
          sysctl -w net.ipv4.conf.macvlan0.rp_filter=0
          
          # 确保 Forward 链是 ACCEPT
          iptables -P FORWARD ACCEPT
          
          # 1. 保存 OVN 网络原本的网关
          OVN_GW=$(ip route show default | awk '/default/ {print $3}')
          
          # 2. 为 Pod 网络和 Service 网络添加路由，确保走 eth0
          #    如果您知道具体 CIDR，请替换 10.0.0.0/8 和 172.16.0.0/12
          ip route add 10.0.0.0/8 via $OVN_GW dev eth0
          ip route add 172.16.0.0/12 via $OVN_GW dev eth0
          
          # 3. 确保 macvlan0 接口已启动
          ip link set macvlan0 up
          
          # 4. 切换默认路由指向物理网关
          # 这一步非常重要，否则 Pod 里的包不知道怎么去外网
          ip route del default
          ip route add default via 192.168.99.1 dev macvlan0
          
          # 5. 设置核心 SNAT 规则
          iptables -t nat -A POSTROUTING -o macvlan0 -j SNAT --to-source 192.168.99.2
          
          # 5. 为了调试方便，加一条日志
          echo "交大版手动网关已启动！出口 IP: 192.168.99.2，物理网关: 192.168.99.1";
          
          # 6. 防止容器退出
          sleep infinity;
        securityContext:
          privileged: true
EOF


```

## OVN & Node Network Patches

Required configurations to bypass default OVN security checks and allow the gateway implementation to function.

Note: OVN operations now require executing commands on the specific `ovnkube-node` pod running on the same node as the target resource, due to the distributed nature of the OVN database in newer versions.

### 1. Variables & Helper Functions
Define a reusable function to apply OVN patches (Port Security clearance and ACLs) to any target pod.

```bash
# Helper: Execute OVN command on the specific node
apply_ovn_cmd() {
    local NODE=$1
    local CMD=$2
    local OVN_POD=$(oc get pods -n openshift-ovn-kubernetes --field-selector spec.nodeName=$NODE -l app=ovnkube-node -o jsonpath='{.items[0].metadata.name}')
    echo "  [Exec] Node: $NODE (Pod: $OVN_POD)"
    oc exec -n openshift-ovn-kubernetes $OVN_POD -c ovn-controller -- bash -c "$CMD"
}

# Core Logic: Patch OVN for a specific Pod
patch_ovn_pod() {
    local NS=$1
    local LABEL=$2
    
    echo "=================================================="
    echo "Configuring Pod [$LABEL] in Namespace [$NS]"
    
    # 1. Wait for Pod
    oc wait --for=condition=Ready pod -l $LABEL -n $NS --timeout=60s > /dev/null
    
    # 2. Get Details
    local POD_NAME=$(oc get pod -n $NS -l $LABEL -o jsonpath='{.items[0].metadata.name}')
    local LSP_NAME="${NS}_${POD_NAME}"
    local NODE_NAME=$(oc get pod -n $NS $POD_NAME -o jsonpath='{.spec.nodeName}')
    
    echo "  Target: Pod=$POD_NAME | Node=$NODE_NAME | LSP=$LSP_NAME"
    
    # 3. Clear Port Security
    # Allow traffic with arbitrary IPs (essential for routing/return traffic)
    echo "  >> Clearing Port Security..."
    local CMD_CLEAR="ovn-nbctl clear Logical_Switch_Port $LSP_NAME port_security"
    apply_ovn_cmd "$NODE_NAME" "$CMD_CLEAR"
    
    # 4. Aggressive ACL Cleanup & Setup
    # Search and remove all ACLs related to this LSP with priorities 31000 and 32000 to ensure a clean slate.
    echo "  >> Cleaning up old ACLs (Priority 31000 and 32000) for $LSP_NAME..."
    # Fix: use 'find' for priority (supported) then grep for LSP name (since match~= is not supported directly in find)
    local CMD_CLEAN="for p in 31000 32000; do \
      for u in \$(ovn-nbctl --format=csv --no-heading --columns=_uuid,match find ACL priority=\$p | grep \"$LSP_NAME\" | awk -F, '{print \$1}'); do \
        echo \"    Removing ACL \$u\"; \
        ovn-nbctl remove Logical_Switch $NODE_NAME acls \$u; \
      done; \
    done"
    apply_ovn_cmd "$NODE_NAME" "$CMD_CLEAN"

    # 5. Add Stateless ACLs (Priority 31000)
    # from-lport: Allow Egress (Pod -> Switch)
    echo "  >> Adding 'from-lport' stateless allow rule..."
    local CMD_ACL_FROM="ovn-nbctl --type=switch acl-add $NODE_NAME from-lport 31000 \"inport == \\\"$LSP_NAME\\\"\" allow-stateless"
    apply_ovn_cmd "$NODE_NAME" "$CMD_ACL_FROM"

    # to-lport: Allow Ingress (Switch -> Pod)
    echo "  >> Adding 'to-lport' stateless allow rule..."
    local CMD_ACL_TO="ovn-nbctl --type=switch acl-add $NODE_NAME to-lport 31000 \"outport == \\\"$LSP_NAME\\\"\" allow-stateless"
    apply_ovn_cmd "$NODE_NAME" "$CMD_ACL_TO"

    # 6. Add Stateful ACLs (Priority 32000)
    # These rules allow related traffic for Service and Cluster networks, bypassing the default drop behavior.
    echo "  >> Adding Stateful ACLs (Priority 32000)..."
    SERVICE_CIDR="172.22.0.0/16"
    CLUSTER_CIDR="10.132.0.0/14"

    # Egress Rules
    CMD_ACL_FROM="ovn-nbctl --type=switch acl-add $NODE_NAME from-lport 32000 \"inport == \\\"$LSP_NAME\\\" && ip4.dst == $SERVICE_CIDR\" allow"
    apply_ovn_cmd "$NODE_NAME" "$CMD_ACL_FROM"

    CMD_ACL_FROM="ovn-nbctl --type=switch acl-add $NODE_NAME from-lport 32000 \"inport == \\\"$LSP_NAME\\\" && ip4.dst == $CLUSTER_CIDR\" allow"
    apply_ovn_cmd "$NODE_NAME" "$CMD_ACL_FROM"
    
    # Ingress Rules (Return Traffic)
    CMD_ACL_TO="ovn-nbctl --type=switch acl-add $NODE_NAME to-lport 32000 \"outport == \\\"$LSP_NAME\\\" && ip4.src == $SERVICE_CIDR\" allow"
    apply_ovn_cmd "$NODE_NAME" "$CMD_ACL_TO"
    
    CMD_ACL_TO="ovn-nbctl --type=switch acl-add $NODE_NAME to-lport 32000 \"outport == \\\"$LSP_NAME\\\" && ip4.src == $CLUSTER_CIDR\" allow"
    apply_ovn_cmd "$NODE_NAME" "$CMD_ACL_TO"
    
    echo "  Done."
}
```

### 2. Apply OVN Patches
Execute the patching function for both the Gateway Pod and the Business Pod.

```bash
# 1. Patch Gateway Pod
GW_POD_LABEL="app=ns-blue-gateway"
GW_NAMESPACE="ns-egress-infra"
patch_ovn_pod "$GW_NAMESPACE" "$GW_POD_LABEL"

# 2. Patch Business Pod
# Ensure the business pod also bypasses OVN checks for return traffic
BUS_POD_LABEL="app=business-app"
BUS_NAMESPACE="ns-blue"
patch_ovn_pod "$BUS_NAMESPACE" "$BUS_POD_LABEL"
```

```bash
# Helper function to inspect OVN config for a pod
inspect_ovn_config() {
    local NS=$1
    local LABEL=$2
    
    echo "Finding pod details..."
    # Get Kubernetes Pod info
    local POD_NAME=$(oc get pod -n $NS -l $LABEL --no-headers -o custom-columns=":metadata.name" | head -n 1)
    
    if [ -z "$POD_NAME" ]; then
        echo "ERROR: Pod not found for $NS / $LABEL"
        return
    fi

    local NODE=$(oc get pod -n $NS $POD_NAME -o jsonpath='{.spec.nodeName}')
    local LSP_NAME="${NS}_${POD_NAME}"
    
    # Find local OVN pod for executing commands
    local OVN_POD=$(oc get pods -n openshift-ovn-kubernetes --field-selector spec.nodeName=$NODE -l app=ovnkube-node -o jsonpath='{.items[0].metadata.name}')

    echo "================================================================================"
    echo "TARGET: $POD_NAME"
    echo "LSP:    $LSP_NAME"
    echo "NODE:   $NODE (Exec via $OVN_POD)"
    echo "================================================================================"

    echo ">>> [1] Checking Port Security (Expect: [])"
    oc exec -n openshift-ovn-kubernetes $OVN_POD -c ovn-controller -- \
        ovn-nbctl list Logical_Switch_Port "$LSP_NAME" | grep -E "name|port_security"

    echo ""
    echo ">>> [2] Checking Stateless ACLs (Expect: Priority 31000 allow-stateless)"
    # We grep for the port name in the ACL list to find relevant rules
    oc exec -n openshift-ovn-kubernetes $OVN_POD -c ovn-controller -- \
        ovn-nbctl list ACL | grep -C 5 "$LSP_NAME" | grep -E "priority|match|action|direction"
    echo ""
}

# Run inspection for both

inspect_ovn_config "$BUS_NAMESPACE" "$BUS_POD_LABEL"
```

### automatically controller

```python
# 文件名: controller.py
import os
import time
import threading
from kubernetes import client, config, watch

# 配置常量
# 配置常量
NAMESPACE = "ns-blue"
GATEWAY_NAMESPACE = "ns-egress-infra"
APB_NAME = "ns-blue-route"
GATEWAY_LABEL = "app=ns-blue-gateway"
GROUP = "k8s.ovn.org"
VERSION = "v1"
PLURAL = "adminpolicybasedexternalroutes"

def get_current_gateway_ips(v1):
    """获取当前所有 Running 状态的网关 Pod IP"""
    try:
        pods = v1.list_namespaced_pod(GATEWAY_NAMESPACE, label_selector=GATEWAY_LABEL)
        return sorted([p.status.pod_ip for p in pods.items if p.status.phase == "Running" and p.status.pod_ip])
    except Exception as e:
        print(f"Error listing pods: {e}")
        return []

def reconcile_apb():
    """核心调谐逻辑：对比并强制同步"""
    config.load_incluster_config()
    v1 = client.CoreV1Api()
    api = client.CustomObjectsApi()

    # 1. 获取当前网关真实的 Pod IP 列表 (期望状态)
    desired_ips = get_current_gateway_ips(v1)
    if not desired_ips:
        print("No running gateway pods found. Skipping sync.")
        return

    desired_formatted = [{"ip": ip} for ip in desired_ips]

    try:
        # 2. 读取当前 APB 资源的实际状态
        current_apb = api.get_cluster_custom_object(GROUP, VERSION, PLURAL, APB_NAME)
        current_static_hops = current_apb.get("spec", {}).get("nextHops", {}).get("static", [])

        # 3. 幂等性检查：对比期望 IP 和 实际 IP
        # 提取当前 IP 字符串进行对比
        current_ips_str = sorted([hop["ip"] for hop in current_static_hops if "ip" in hop])
        
        if current_ips_str == desired_ips:
            # 如果完全一致，直接退出，防止触发 Watch 的死循环
            return 

        # 4. 如果不一致，执行强制修正
        print(f"Detect deviation! Desired: {desired_ips}, Current in APB: {current_ips_str}. Correcting...")
        body = {"spec": {"nextHops": {"static": desired_formatted}}}
        api.patch_cluster_custom_object(GROUP, VERSION, PLURAL, APB_NAME, body)
        print("APB Successfully enforced.")

    except Exception as e:
        print(f"Reconciliation failed: {e}")

def watch_pods():
    """监听 Pod 变化"""
    v1 = client.CoreV1Api()
    w = watch.Watch()
    print("Started watching Pods...")
    for event in w.stream(v1.list_namespaced_pod, namespace=GATEWAY_NAMESPACE, label_selector=GATEWAY_LABEL):
        # 只要 Pod 状态有变（新增、删除、重启），执行调谐
        reconcile_apb()

def watch_apb():
    """监听 APB 资源变化（防止人为手动修改）"""
    api = client.CustomObjectsApi()
    w = watch.Watch()
    print("Started watching APB Resource...")
    # 注意：watch 的资源必须对应其 plural 名称
    for event in w.stream(api.list_cluster_custom_object, GROUP, VERSION, PLURAL):
        resource = event.get('object', {})
        if resource.get('metadata', {}).get('name') == APB_NAME:
            # 只要 APB 被修改（不管是人为还是程序），执行调谐
            # reconcile_apb 内部的对比逻辑会防止死循环
            reconcile_apb()

if __name__ == "__main__":
    config.load_incluster_config()
    
    # 启动时先执行一次强制同步
    reconcile_apb()

    # 使用多线程同时监听两类资源
    t1 = threading.Thread(target=watch_pods)
    t2 = threading.Thread(target=watch_apb)

    t1.start()
    t2.start()

    t1.join()
    t2.join()
```
