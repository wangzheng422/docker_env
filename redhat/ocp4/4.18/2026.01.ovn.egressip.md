# OVN egressip limitation workaround

## Background

customer install a ocp 4.18 on 3rd part osp, using baremetal install method. and the osp has limitation of 10 elastic ip on each node. but the security require each namespace has a dedicate egress ip, and there are 40+ namespaces. and the elastic ip pools contains different subnets, and worker node has different associated relationship with elastic ip pools subnets. current ovn does not support node selector for egressip. and only single pool of egressip is supported. so we need to implement a workaround to support this requirement.

the basic idea is to use a gateway pod for each namespace to provide egress ip for the namespace. use ovn config to route the traffic to the gateway pod for the namespace pod. and use a tool pod to monitor the gateway pod's ip, and change ovn config's next hop ip when the gateway pod restart.

## Workaround

```bash

oc create ns ns-blue

# 创建 SA 
oc create sa apb-syncer -n ns-blue

# 定义权限：允许读取 Pod 并修改 APB
cat <<EOF | oc apply -f -
apiVersion: rbac.authorization.k8s.io/v1
kind: Role
metadata:
  name: apb-syncer-role
  namespace: ns-blue
rules:
- apiGroups: [""]
  resources: ["pods"]
  verbs: ["get", "list", "watch"]
- apiGroups: ["k8s.ovn.org"]
  resources: ["adminpolicybasedexternalroutes"]
  verbs: ["get", "list", "patch", "update"]
EOF

# 绑定权限
oc adm policy add-role-to-user apb-syncer-role -z apb-syncer -n ns-blue


oc label node worker-01-demo egress-node=ns-blue-group


cat <<EOF | oc apply -f -
apiVersion: apps/v1
kind: Deployment
metadata:
  name: ns-blue-gateway
  namespace: ns-blue
spec:
  replicas: 1 
  selector:
    matchLabels:
      app: ns-blue-gateway
  template:
    metadata:
      labels:
        app: ns-blue-gateway
      annotations:
        # 启用 macvlan
        pod.network.openshift.io/assign-macvlan: "true"
    spec:
      nodeSelector:
        egress-node: ns-blue-group
      containers:
      - name: gateway
        image: registry.redhat.io/openshift4/ose-egress-router:latest
        env:
        - name: EGRESS_SOURCE
          value: 192.168.99.2       # 固定的出口 EIP
        - name: EGRESS_GATEWAY
          value: 192.168.99.1       # 物理网关地址
        - name: EGRESS_ROUTER_MODE
          value: init
        securityContext:
          privileged: true     # 必须开启特权模式以操作 iptables
EOF

```

```python
# 文件名: controller.py
import os
from kubernetes import client, config, watch

def update_apb(ips):
    api = client.CustomObjectsApi()
    body = {"spec": {"nextHops": {"static": ips}}}
    try:
        api.patch_namespaced_custom_object("k8s.ovn.org", "v1", "ns-blue", "adminpolicybasedexternalroutes", "ns1-route", body)
        print(f"APB Updated: {ips}")
    except Exception as e:
        print(f"Error: {e}")

def main():
    config.load_incluster_config()
    v1 = client.CoreV1Api()
    w = watch.Watch()
    for event in w.stream(v1.list_namespaced_pod, namespace="ns-blue", label_selector="app=ns-blue-gateway"):
        pods = v1.list_namespaced_pod("ns-blue", label_selector="app=ns-blue-gateway")
        ips = [p.status.pod_ip for p in pods.items if p.status.phase == "Running" and p.status.pod_ip]
        if ips:
            update_apb(ips)

if __name__ == "__main__":
    main()
```

```bash

# 将脚本存入 ConfigMap
oc create configmap apb-script --from-file=controller.py -n ns-blue

# 部署 Controller
cat <<EOF | oc apply -f -
apiVersion: apps/v1
kind: Deployment
metadata:
  name: apb-controller
  namespace: ns-blue
spec:
  replicas: 1
  template:
    metadata:
      labels:
        app: apb-controller
    spec:
      serviceAccountName: apb-syncer
      containers:
      - name: controller
        image: registry.access.redhat.com/ubi9/python-312
        command: ["/bin/sh", "-c", "pip install kubernetes --user && python /mnt/controller.py"]
        volumeMounts:
        - name: script
          mountPath: /mnt
      volumes:
      - name: script
        configmap:
          name: apb-script
EOF


cat <<EOF | oc apply -f -
apiVersion: k8s.ovn.org/v1
kind: AdminPolicyBasedExternalRoute
metadata:
  name: ns1-route
  namespace: ns-blue
spec:
  from:
    namespaceSelector:
      matchLabels:
        kubernetes.io/metadata.name: ns-blue
  nextHops:
    static:
    - 0.0.0.0 # 初始占位，将被 Controller 更新
EOF


```
