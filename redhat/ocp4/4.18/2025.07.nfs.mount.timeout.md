`nfs-setup.sh`

```bash
#!/bin/bash

# NFS annd base path for the test
NFS_BASE_PATH="/srv/nfs/pressure-test"

echo "Creating and setting permissions for the main export directory: ${NFS_BASE_PATH}"
sudo mkdir -p ${NFS_BASE_PATH}
sudo chmod 777 ${NFS_BASE_PATH}
sudo chown nobody:nobody ${NFS_BASE_PATH}

# Add a SINGLE export entry for the parent directory to /etc/exports
# This allows any client (*) to mount this directory and its subdirectories.
# We'll first remove any old entries for this path to avoid duplicates.
sudo sed -i "\|${NFS_BASE_PATH}|d" /etc/exports
echo "${NFS_BASE_PATH} *(rw,sync,no_root_squash)" | sudo tee -a /etc/exports

# Create the 500 unique subdirectories that the PVs will point to
echo "Creating 500 subdirectories..."
for i in $(seq 1 500); do
  # Format directory name, e.g., pv-001, pv-002 ...
  DIR_NAME=$(printf "pv-%03d" ${i})
  FULL_PATH="${NFS_BASE_PATH}/${DIR_NAME}"

  # The 'if' condition makes the script re-runnable without errors
  if [ ! -d "${FULL_PATH}" ]; then
    echo "Creating directory: ${FULL_PATH}"
    sudo mkdir -p ${FULL_PATH}
    sudo chmod 777 ${FULL_PATH}
    sudo chown nobody:nobody ${FULL_PATH}
  fi
done

# Reload NFS exports to apply the changes from /etc/exports
echo "Reloading NFS exports..."
sudo exportfs -ra

echo "NFS server setup complete. One parent directory exported, 500 subdirectories are ready."

```

`create-nfs-pressure-test.sh`

```bash
#!/bin/bash

# --- 配置变量 ---
# 请将这些值替换为您环境的实际值
NFS_SERVER_IP="192.168.99.1"
NFS_BASE_PATH="/srv/nfs/pressure-test"
NAMESPACE="nfs-test-demo"
STORAGE_CLASS_NAME="nfs-retain-sc" # 自定义 StorageClass 名称
POD_COUNT=500

# --- 脚本开始 ---

echo "Creating project ${NAMESPACE}..."
oc new-project ${NAMESPACE} || true # 如果项目已存在则忽略错误

# 1. 创建一个自定义的 StorageClass
# 这个 SC 用于手动绑定的 PV，设置 reclaimPolicy 为 Retain
echo "Creating StorageClass..."
cat <<EOF | oc apply -f -
apiVersion: storage.k8s.io/v1
kind: StorageClass
metadata:
  name: ${STORAGE_CLASS_NAME}
provisioner: kubernetes.io/no-provisioner
volumeBindingMode: Immediate
reclaimPolicy: Retain
EOF

echo "Starting to create ${POD_COUNT} PVs, PVCs, and Pods..."

# 2. 循环创建所有资源
for i in $(seq 1 ${POD_COUNT}); do
  # 格式化名称
  SUFFIX=$(printf "%03d" ${i})
  PV_NAME="nfs-pv-${SUFFIX}"
  PVC_NAME="nfs-pvc-${SUFFIX}"
  POD_NAME="nfs-pod-${SUFFIX}"
  NFS_PATH="${NFS_BASE_PATH}/pv-${SUFFIX}"

  echo "--- Creating resources for set ${SUFFIX} ---"

  # 生成并应用 PV 的 YAML
  cat <<EOF | oc apply -f -
apiVersion: v1
kind: PersistentVolume
metadata:
  name: ${PV_NAME}
spec:
  capacity:
    storage: 1Mi # 容量可以很小，因为我们只测试挂载
  accessModes:
    - ReadWriteMany
  storageClassName: ${STORAGE_CLASS_NAME}
  persistentVolumeReclaimPolicy: Retain
  nfs:
    server: "${NFS_SERVER_IP}"
    path: "${NFS_PATH}"
EOF

  # 生成并应用 PVC 的 YAML
  cat <<EOF | oc apply -n ${NAMESPACE} -f -
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: ${PVC_NAME}
spec:
  accessModes:
    - ReadWriteMany
  storageClassName: ${STORAGE_CLASS_NAME}
  resources:
    requests:
      storage: 1Mi
  volumeName: ${PV_NAME}
EOF

  # 生成并应用 Pod 的 YAML
  # Pod 启动后会简单地检查挂载并保持运行
  cat <<EOF | oc apply -n ${NAMESPACE} -f -
apiVersion: v1
kind: Pod
metadata:
  name: ${POD_NAME}
spec:
  containers:
  - name: test-container
    image: registry.access.redhat.com/ubi8/ubi-minimal
    command: ["/bin/sh", "-c"]
    args:
      - |
        echo "Pod ${POD_NAME} is running and will start writing data."
        LOG_FILE="/data/activity.log"
        
        # This loop will run forever
        while true; do
          # Get the current timestamp
          TIMESTAMP=\$(date)
          
          # Append the timestamp and a message to the log file
          echo "\${TIMESTAMP}: Pod ${POD_NAME} is writing." > \${LOG_FILE}
          
          # Wait for one second
          sleep 0.1
        done
    volumeMounts:
    - name: nfs-volume
      mountPath: /data
  volumes:
  - name: nfs-volume
    persistentVolumeClaim:
      claimName: ${PVC_NAME}
EOF

done

echo "All ${POD_COUNT} sets of resources have been created."
```
