# OVN egressip limitation workaround

## Background

customer install a ocp 4.18 on 3rd part osp, using baremetal install method. and the osp has limitation of 10 elastic ip on each node. but the security require each namespace has a dedicate egress ip, and there are 40+ namespaces. and the elastic ip pools contains different subnets, and worker node has different associated relationship with elastic ip pools subnets. current ovn does not support node selector for egressip. and only single pool of egressip is supported. so we need to implement a workaround to support this requirement.

the basic idea is to use a gateway pod for each namespace to provide egress ip for the namespace. use ovn config to route the traffic to the gateway pod for the namespace pod. and use a tool pod to monitor the gateway pod's ip, and change ovn config's next hop ip when the gateway pod restart.

in this demo, we have ocp 4.18 installed, 3 master, 2 worker, worker node ip is 192.168.99.26, 192.168.99.27, I have a big rhel 9 machine, and I create the 5 node with kvm, so there should no mac restrict on the network.

## OVN & Node Network Patches

Required configurations to bypass default OVN security checks and allow the gateway implementation to function.

Note: OVN operations now require executing commands on the specific `ovnkube-node` pod running on the same node as the target resource, due to the distributed nature of the OVN database in newer versions.

### 1. Variables & Helper Functions
Define a reusable function to apply OVN patches (Port Security clearance and ACLs) to any target pod.

```bash
# Helper: Execute OVN command on the specific node
apply_ovn_cmd() {
    local NODE=$1
    local CMD=$2
    local OVN_POD=$(oc get pods -n openshift-ovn-kubernetes --field-selector spec.nodeName=$NODE -l app=ovnkube-node -o jsonpath='{.items[0].metadata.name}')
    echo "  [Exec] Node: $NODE (Pod: $OVN_POD)"
    oc exec -n openshift-ovn-kubernetes $OVN_POD -c ovn-controller -- bash -c "$CMD"
}

# Core Logic: Clean OVN settings for a specific Pod
clean_ovn_pod() {
    local NS=$1
    local LABEL=$2
    
    echo "=================================================="
    echo "Cleaning Pod [$LABEL] in Namespace [$NS]"
    
    # 1. Wait for Pod
    oc wait --for=condition=Ready pod -l $LABEL -n $NS --timeout=60s > /dev/null
    
    # 2. Get Details
    local POD_NAME=$(oc get pod -n $NS -l $LABEL -o jsonpath='{.items[0].metadata.name}')
    local LSP_NAME="${NS}_${POD_NAME}"
    local NODE_NAME=$(oc get pod -n $NS $POD_NAME -o jsonpath='{.spec.nodeName}')
    
    echo "  Target: Pod=$POD_NAME | Node=$NODE_NAME | LSP=$LSP_NAME"
    
    # 3. Aggressive ACL Cleanup - using custom priority 31821
    echo "  >> Cleaning up old ACLs (Priority 31821) for $LSP_NAME..."
    local CMD_CLEAN="for u in \$(ovn-nbctl --format=csv --no-heading --columns=_uuid,match find ACL priority=31821 | grep \"$LSP_NAME\" | awk -F, '{print \$1}'); do \
        echo \"    Removing ACL \$u\"; \
        ovn-nbctl remove Logical_Switch $NODE_NAME acls \$u; \
      done"
    apply_ovn_cmd "$NODE_NAME" "$CMD_CLEAN"
    
    # 5. Clean up problematic global ACL rules that affect API server connectivity
    echo "  >> Cleaning up ALL ACLs with Priority 31821..."
    local CMD_CLEAN="for u in \$(ovn-nbctl --format=csv --no-heading --columns=_uuid find ACL priority=31821 | awk -F, '{print \$1}'); do \
        echo \"    Removing ACL \$u\"; \
        ovn-nbctl remove Logical_Switch $NODE_NAME acls \$u; \
      done"
    apply_ovn_cmd "$NODE_NAME" "$CMD_CLEAN"
    
    echo "  Clean Done."
}



# Core Logic: Apply OVN patches for a specific Pod
# Usage: patch_ovn_pod <namespace> <label> [port_security_mode]
# Parameters:
#   - port_security_mode: "clear_port_security" or "keep_port_security" (default: "keep_port_security")
#     * Gateway Pod: use "clear_port_security" (needs to forward traffic with different source IPs)
#     * Business Pod: use "keep_port_security" (only sends its own traffic, keep security)
patch_ovn_pod() {
    local NS=$1
    local LABEL=$2
    local PORT_SECURITY_MODE=${3:-keep_port_security}  # Default: keep_port_security (enhanced security)
    
    echo "=================================================="
    echo "Configuring Pod [$LABEL] in Namespace [$NS]"
    echo "  Port Security Mode: $PORT_SECURITY_MODE"
    
    # 1. Wait for Pod
    oc wait --for=condition=Ready pod -l $LABEL -n $NS --timeout=60s > /dev/null
    
    # 2. Get Details
    local POD_NAME=$(oc get pod -n $NS -l $LABEL -o jsonpath='{.items[0].metadata.name}')
    local LSP_NAME="${NS}_${POD_NAME}"
    local NODE_NAME=$(oc get pod -n $NS $POD_NAME -o jsonpath='{.spec.nodeName}')
    
    echo "  Target: Pod=$POD_NAME | Node=$NODE_NAME | LSP=$LSP_NAME"
    
    # 3. Clear Port Security (conditional)
    if [ "$PORT_SECURITY_MODE" = "clear_port_security" ]; then
        echo "  >> Clearing Port Security (Gateway mode - allows forwarding)..."
        local CMD_CLEAR="ovn-nbctl clear Logical_Switch_Port $LSP_NAME port_security"
        apply_ovn_cmd "$NODE_NAME" "$CMD_CLEAR"
    else
        echo "  >> Keeping Port Security (Business Pod mode - enhanced security)"
    fi
    
    # 4. Add Stateless ACLs for ALL traffic (Priority 31821)
    # CRITICAL: All traffic (TCP and UDP) uses stateless processing
    # This is necessary because return traffic from external IPs (8.8.8.8, etc.)
    # would be blocked by stateful ACLs that check source IP
    # from-lport: Allow ALL Egress (Pod -> Switch)
    echo "  >> Adding 'from-lport' stateless allow rule for ALL traffic (Priority 31821)..."
    local CMD_ACL_FROM="ovn-nbctl --type=switch acl-add $NODE_NAME from-lport 31821 \"inport == \\\"$LSP_NAME\\\"\" allow-stateless"
    apply_ovn_cmd "$NODE_NAME" "$CMD_ACL_FROM"

    # to-lport: Allow ALL Ingress (Switch -> Pod)
    # This allows return traffic from ANY external IP to bypass source IP checks
    echo "  >> Adding 'to-lport' stateless allow rule for ALL traffic (Priority 31821)..."
    local CMD_ACL_TO="ovn-nbctl --type=switch acl-add $NODE_NAME to-lport 31821 \"outport == \\\"$LSP_NAME\\\"\" allow-stateless"
    apply_ovn_cmd "$NODE_NAME" "$CMD_ACL_TO"
    
    echo "  Done."
}
```

```bash
# Core Logic: Clean all ACLs with specific priority on a node
# Usage: clean_node_acls_by_priority <node_name> <priority>
# Parameters:
#   - node_name: Target node name (e.g., worker-01-demo)
#   - priority: ACL priority to clean (e.g., 31821, 32831)
# Example: clean_node_acls_by_priority worker-01-demo 31821
clean_node_acls_by_priority() {
    local NODE_NAME=$1
    local PRIORITY=$2
    
    if [ -z "$NODE_NAME" ] || [ -z "$PRIORITY" ]; then
        echo "Usage: clean_node_acls_by_priority <node_name> <priority>"
        return 1
    fi
    
    echo "=================================================="
    echo "Cleaning ACLs on Node [$NODE_NAME] with Priority [$PRIORITY]"
    
    # Get OVN pod for this node
    local OVN_POD=$(oc get pods -n openshift-ovn-kubernetes --field-selector spec.nodeName=$NODE_NAME -l app=ovnkube-node -o jsonpath='{.items[0].metadata.name}')
    
    if [ -z "$OVN_POD" ]; then
        echo "  [ERROR] No ovnkube-node pod found on node $NODE_NAME"
        return 1
    fi
    
    echo "  [Info] Using OVN Pod: $OVN_POD"
    
    # List all ACLs with the specified priority
    echo "  >> Listing ACLs with priority $PRIORITY..."
    local CMD_LIST="ovn-nbctl --format=csv --no-heading --columns=_uuid,match find ACL priority=$PRIORITY"
    local ACL_LIST=$(oc exec -n openshift-ovn-kubernetes $OVN_POD -c ovn-controller -- bash -c "$CMD_LIST" 2>/dev/null)
    
    if [ -z "$ACL_LIST" ]; then
        echo "  >> No ACLs found with priority $PRIORITY"
        echo "  Clean Done."
        return 0
    fi
    
    # Count ACLs
    local ACL_COUNT=$(echo "$ACL_LIST" | wc -l)
    echo "  >> Found $ACL_COUNT ACL(s) with priority $PRIORITY"
    
    # Delete all ACLs with this priority
    echo "  >> Removing all ACLs with priority $PRIORITY..."
    local CMD_CLEAN="for u in \$(ovn-nbctl --format=csv --no-heading --columns=_uuid find ACL priority=$PRIORITY | awk -F, '{print \$1}'); do \
        echo \"    Removing ACL \$u\"; \
        ovn-nbctl remove Logical_Switch $NODE_NAME acls \$u; \
      done"
    apply_ovn_cmd "$NODE_NAME" "$CMD_CLEAN"
    
    echo "  Clean Done."
}
```

### automatically controller

```python
# 文件名: controller.py
import os
import time
import threading
from kubernetes import client, config, watch

# 配置常量
# 配置常量
NAMESPACE = "ns-blue"
GATEWAY_NAMESPACE = "ns-egress-infra"
APB_NAME = "ns-blue-route"
GATEWAY_LABEL = "app=ns-blue-gateway"
GROUP = "k8s.ovn.org"
VERSION = "v1"
PLURAL = "adminpolicybasedexternalroutes"

def get_current_gateway_ips(v1):
    """获取当前所有 Running 状态的网关 Pod IP"""
    try:
        pods = v1.list_namespaced_pod(GATEWAY_NAMESPACE, label_selector=GATEWAY_LABEL)
        return sorted([p.status.pod_ip for p in pods.items if p.status.phase == "Running" and p.status.pod_ip])
    except Exception as e:
        print(f"Error listing pods: {e}")
        return []

def reconcile_apb():
    """核心调谐逻辑：对比并强制同步"""
    config.load_incluster_config()
    v1 = client.CoreV1Api()
    api = client.CustomObjectsApi()

    # 1. 获取当前网关真实的 Pod IP 列表 (期望状态)
    desired_ips = get_current_gateway_ips(v1)
    if not desired_ips:
        print("No running gateway pods found. Skipping sync.")
        return

    desired_formatted = [{"ip": ip} for ip in desired_ips]

    try:
        # 2. 读取当前 APB 资源的实际状态
        current_apb = api.get_cluster_custom_object(GROUP, VERSION, PLURAL, APB_NAME)
        current_static_hops = current_apb.get("spec", {}).get("nextHops", {}).get("static", [])

        # 3. 幂等性检查：对比期望 IP 和 实际 IP
        # 提取当前 IP 字符串进行对比
        current_ips_str = sorted([hop["ip"] for hop in current_static_hops if "ip" in hop])
        
        if current_ips_str == desired_ips:
            # 如果完全一致，直接退出，防止触发 Watch 的死循环
            return 

        # 4. 如果不一致，执行强制修正
        print(f"Detect deviation! Desired: {desired_ips}, Current in APB: {current_ips_str}. Correcting...")
        body = {"spec": {"nextHops": {"static": desired_formatted}}}
        api.patch_cluster_custom_object(GROUP, VERSION, PLURAL, APB_NAME, body)
        print("APB Successfully enforced.")

    except Exception as e:
        print(f"Reconciliation failed: {e}")

def watch_pods():
    """监听 Pod 变化"""
    v1 = client.CoreV1Api()
    w = watch.Watch()
    print("Started watching Pods...")
    for event in w.stream(v1.list_namespaced_pod, namespace=GATEWAY_NAMESPACE, label_selector=GATEWAY_LABEL):
        # 只要 Pod 状态有变（新增、删除、重启），执行调谐
        reconcile_apb()

def watch_apb():
    """监听 APB 资源变化（防止人为手动修改）"""
    api = client.CustomObjectsApi()
    w = watch.Watch()
    print("Started watching APB Resource...")
    # 注意：watch 的资源必须对应其 plural 名称
    for event in w.stream(api.list_cluster_custom_object, GROUP, VERSION, PLURAL):
        resource = event.get('object', {})
        if resource.get('metadata', {}).get('name') == APB_NAME:
            # 只要 APB 被修改（不管是人为还是程序），执行调谐
            # reconcile_apb 内部的对比逻辑会防止死循环
            reconcile_apb()

if __name__ == "__main__":
    config.load_incluster_config()
    
    # 启动时先执行一次强制同步
    reconcile_apb()

    # 使用多线程同时监听两类资源
    t1 = threading.Thread(target=watch_pods)
    t2 = threading.Thread(target=watch_apb)

    t1.start()
    t2.start()

    t1.join()
    t2.join()
```
