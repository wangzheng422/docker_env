# get cert list by content filter


```bash
#!/bin/bash

# ==============================================================================
# 脚本功能: 遍历OpenShift集群所有命名空间下的所有Secret，
#           解码其数据，并判断每个数据项是否为有效的X.509证书。
# ==============================================================================

set -eo pipefail

# 获取所有非系统核心的命名空间列表 (可以修改为您想扫描的范围)
# 获取所有命名空间: oc get ns -o jsonpath='{.items[*].metadata.name}'
# 这里排除了一些常见的、庞大的操作员命名空间以加快速度，您可以按需调整
NAMESPACES=$(oc get ns -o jsonpath='{.items[*].metadata.name}' | xargs -n1 | grep -vE "^(openshift-api-server|openshift-etcd|openshift-sdn|openshift-kni-infra|openshift-kube.*)$")

echo "开始扫描以下命名空间: "
echo "$NAMESPACES"
echo "======================================================================================"
printf "%-40s %-50s %-30s %-10s\n" "NAMESPACE" "SECRET_NAME" "DATA_KEY" "IS_CERT?"
echo "======================================================================================"

# 遍历每一个命名空间
for ns in $NAMESPACES; do
    # 获取该命名空间下所有secret的JSON输出
    SECRETS_JSON=$(oc get secret -n "$ns" -o json)

    # 使用jq遍历每一个secret
    echo "$SECRETS_JSON" | jq -c '.items[] | {name: .metadata.name, data: .data}' | while read -r secret_line; do
        SECRET_NAME=$(echo "$secret_line" | jq -r '.name')
        
        # 检查secret是否有.data字段
        if ! echo "$secret_line" | jq -e '.data' > /dev/null; then
            continue
        fi

        # 遍历.data字段下的所有key
        echo "$secret_line" | jq -r '.data | keys[]' | while read -r key; do
            # 提取并解码value
            # 使用--arg来安全传递变量给jq
            DECODED_DATA=$(echo "$secret_line" | jq -r --arg k "$key" '.data[$k]' | base64 -d 2>/dev/null)
            
            IS_CERT="No"
            # 检查解码后的数据是否以'-----BEGIN CERTIFICATE-----'开头
            # 并用openssl进行最终验证
            if [[ "$DECODED_DATA" == *"-----BEGIN CERTIFICATE-----"* ]]; then
                # 使用openssl来验证它是否是一个真正有效的X.509证书
                # -noout: 不输出证书内容
                # -text: 解析证书 (如果失败，命令会返回非零退出码)
                if echo "$DECODED_DATA" | openssl x509 -noout -text > /dev/null 2>&1; then
                    IS_CERT="Yes"
                fi
            fi
            
            # 如果是证书，就打印出来
            if [ "$IS_CERT" == "Yes" ]; then
                 printf "%-40s %-50s %-30s %-10s\n" "$ns" "$SECRET_NAME" "$key" "$IS_CERT"
            fi
        done
    done
done

echo "======================================================================================"
echo "扫描完成。"
```
