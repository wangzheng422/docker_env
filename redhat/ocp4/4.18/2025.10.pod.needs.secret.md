# list pod that require secret

customer always want a complete pod list, that is affected by tls secret rotation. we will try to list that.

The basic idea is, if pod restart when tls secret rotation, that pod is controlled by ocp operator, but operator restart pod, because pod access tls secret content, and operator restart pod just force the pod to re-read the tls secret or other config.

So, we just need to list pod yaml, and scan for it relate to secret, that is all

```bash

echo "开始进行非破坏性 Secret 依赖分析..."
echo "================================================================================"

# 获取所有类型为 kubernetes.io/tls 或 Opaque 的 Secret，格式为 "命名空间 Secret名称"
# 修正了 jsonpath 过滤器中的 OR 运算符 (||)
oc get secret -A \
  -o jsonpath='{range .items[?(@.type=="kubernetes.io/tls")]}{.metadata.namespace}{" "}{.metadata.name}{"\n"}{end}{range .items[?(@.type=="Opaque")]}{.metadata.namespace}{" "}{.metadata.name}{"\n"}{end}' | \
sort -u | \
while read -r SECRET_NAMESPACE SECRET_NAME; do
  
  echo "[INFO] 正在检查 Secret: '$SECRET_NAME' (命名空间: '$SECRET_NAMESPACE')"

  # 在该命名空间中查找所有引用了此 Secret 的 Pod
AFFECTED_PODS=$(oc get pods -n "$SECRET_NAMESPACE" -o json | jq -r --arg SECRET_NAME "$SECRET_NAME" '
  .items[] |
    select(
      # 条件1: 检查 volumes
      (any(.spec.volumes[]?; .secret?.secretName == $SECRET_NAME)) or

      # 条件2: 检查 envFrom
      (any(.spec.containers[]?; any(.envFrom[]?; .secretRef?.name == $SECRET_NAME))) or

      # 条件3: 检查 env
      (any(.spec.containers[]?; any(.env[]?; .valueFrom?.secretKeyRef?.name == $SECRET_NAME)))
    ) | .metadata.name
')


    echo "  └── 影响以下 Pod:"
    # 将找到的 Pod 列表格式化输出
    echo "$AFFECTED_PODS" | while read -r POD_NAME; do
      echo "      - $POD_NAME"
    done
done
```
