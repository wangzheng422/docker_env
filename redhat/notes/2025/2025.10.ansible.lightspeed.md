```yaml
kind: Secret
apiVersion: v1
metadata:
  name: chatbot-configuration-secret
  namespace: aap
data:
  aap_controller_url: axxxxxx
  aap_gateway_url: axxxx
  chatbot_model: Zxxxx
  chatbot_token: cxxxx
  chatbot_url: axxxxxx
type: Opaque
EOF
```



```yaml
apiVersion: aap.ansible.com/v1alpha1
kind: AnsibleAutomationPlatform

  namespace: aap

  lightspeed:
    chatbot_config_secret_name: chatbot-configuration-secret
```
